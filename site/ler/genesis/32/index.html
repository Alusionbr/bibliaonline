    וַיַּשְׁכֵּם לָבָן בַּבֹּקֶר וַיְנַשֵּׁק לְבָנָיו וְלִבְנוֹתָיו וַיְבָרֶךְ אֶתְהֶם וַיֵּלֶךְ וַיָּשָׁב לָבָן לִמְקֹמֽוֹ

    vayashekem lavan baboqer vayenasheq levanayv velivenotayv vayevarekhe etehem vayelekhe vayashav lavan limeqomo

    —

    2

    וְיַעֲקֹב הָלַךְ לְדַרְכּוֹ וַיִּפְגְּעוּ בוֹ מַלְאֲכֵי אֱלֹהִֽים

    veyaaqov halakhe ledareko vayifegeu vo maleakhey elohiym

    —

    3

    וַיֹּאמֶר יַעֲקֹב כַּאֲשֶׁר רָאָם מַחֲנֵה אֱלֹהִים זֶה וַיִּקְרָא שֵֽׁם הַמָּקוֹם הַהוּא מַֽחֲנָֽיִם

    vayomer yaaqov kaasher raam machaneh elohiym zeh vayiqera shem hamaqom hahu machanayim

    —

    4

    וַיִּשְׁלַח יַעֲקֹב מַלְאָכִים לְפָנָיו אֶל עֵשָׂו אָחִיו אַרְצָה שֵׂעִיר שְׂדֵה אֱדֽוֹם

    vayishelach yaaqov maleakhiym lefanayv el esav achiyv aretsah seiyr sedeh edom

    —

    5

    וַיְצַו אֹתָם לֵאמֹר כֹּה תֹאמְרוּן לַֽאדֹנִי לְעֵשָׂו כֹּה אָמַר עַבְדְּךָ יַעֲקֹב עִם לָבָן גַּרְתִּי וָאֵחַר עַד עָֽתָּה

    vayetsav otam lemor koh tomerun ladoniy leesav koh amar avedekha yaaqov im lavan garetiy vaechar ad atah

    —

    6

    וַֽיְהִי לִי שׁוֹר וַחֲמוֹר צֹאן וְעֶבֶד וְשִׁפְחָה וָֽאֶשְׁלְחָה לְהַגִּיד לַֽאדֹנִי לִמְצֹא חֵן בְּעֵינֶֽיךָ

    vayehiy liy shor vachamor tson veeved veshifechah vaeshelechah lehagiyd ladoniy limetso chen beeyneykha

    —

    7

    וַיָּשֻׁבוּ הַמַּלְאָכִים אֶֽל יַעֲקֹב לֵאמֹר בָּאנוּ אֶל אָחִיךָ אֶל עֵשָׂו וְגַם הֹלֵךְ לִקְרָֽאתְךָ וְאַרְבַּע מֵאוֹת אִישׁ עִמּֽוֹ

    vayashuvu hamaleakhiym el yaaqov lemor banu el achiykha el esav vegam holekhe liqeratekha veareba meot iysh imo

    —

    8

    וַיִּירָא יַעֲקֹב מְאֹד וַיֵּצֶר לוֹ וַיַּחַץ אֶת הָעָם אֲשֶׁר אִתּוֹ וְאֶת הַצֹּאן וְאֶת הַבָּקָר וְהַגְּמַלִּים לִשְׁנֵי מַחֲנֽוֹת

    vayiyra yaaqov meod vayetser lo vayachats et haam asher ito veet hatson veet habaqar vehagemaliym lisheney machanot

    —

    9

    וַיֹּאמֶר אִם יָבוֹא עֵשָׂו אֶל הַמַּחֲנֶה הָאַחַת וְהִכָּהוּ וְהָיָה הַמַּחֲנֶה הַנִּשְׁאָר לִפְלֵיטָֽה

    vayomer im yavo esav el hamachaneh haachat vehikahu vehayah hamachaneh hanishear lifeleytah

    —

    10

    וַיֹּאמֶר יַעֲקֹב אֱלֹהֵי אָבִי אַבְרָהָם וֵאלֹהֵי אָבִי יִצְחָק יְהוָה הָאֹמֵר אֵלַי שׁוּב לְאַרְצְךָ וּלְמוֹלַדְתְּךָ וְאֵיטִיבָה עִמָּֽךְ

    vayomer yaaqov elohey aviy averaham velohey aviy yitsechaq yehvah haomer elay shuv learetsekha ulemoladetekha veeytiyvah imakhe

    —

    11

    קָטֹנְתִּי מִכֹּל הַחֲסָדִים וּמִכָּל הָאֱמֶת אֲשֶׁר עָשִׂיתָ אֶת עַבְדֶּךָ כִּי בְמַקְלִי עָבַרְתִּי אֶת הַיַּרְדֵּן הַזֶּה וְעַתָּה הָיִיתִי לִשְׁנֵי מַחֲנֽוֹת

    qatonetiy mikol hachasadiym umikal haemet asher asiyta et avedekha kiy vemaqeliy avaretiy et hayareden hazeh veatah hayiytiy lisheney machanot

    —

    12

    הַצִּילֵנִי נָא מִיַּד אָחִי מִיַּד עֵשָׂו כִּֽי יָרֵא אָנֹכִי אֹתוֹ פֶּן יָבוֹא וְהִכַּנִי אֵם עַל בָּנִֽים

    hatsiyleniy na miyad achiy miyad esav kiy yare anokhiy oto pen yavo vehikaniy em al baniym

    —

    13

    וְאַתָּה אָמַרְתָּ הֵיטֵב אֵיטִיב עִמָּךְ וְשַׂמְתִּי אֶֽת זַרְעֲךָ כְּחוֹל הַיָּם אֲשֶׁר לֹא יִסָּפֵר מֵרֹֽב

    veatah amareta heytev eytiyv imakhe vesametiy et zareakha kechol hayam asher lo yisafer merov

    —

    14

    וַיָּלֶן שָׁם בַּלַּיְלָה הַהוּא וַיִּקַּח מִן הַבָּא בְיָדוֹ מִנְחָה לְעֵשָׂו אָחִֽיו

    vayalen sham balayelah hahu vayiqach min haba veyado minechah leesav achiyv

    —

    15

    עִזִּים מָאתַיִם וּתְיָשִׁים עֶשְׂרִים רְחֵלִים מָאתַיִם וְאֵילִים עֶשְׂרִֽים

    iziym matayim uteyashiym eseriym recheliym matayim veeyliym eseriym

    —

    16

    גְּמַלִּים מֵינִיקוֹת וּבְנֵיהֶם שְׁלֹשִׁים פָּרוֹת אַרְבָּעִים וּפָרִים עֲשָׂרָה אֲתֹנֹת עֶשְׂרִים וַעְיָרִם עֲשָׂרָֽה

    gemaliym meyniyqot uveneyhem sheloshiym parot arebaiym ufariym asarah atonot eseriym vaeyarim asarah

    —

    17

    וַיִּתֵּן בְּיַד עֲבָדָיו עֵדֶר עֵדֶר לְבַדּוֹ וַיֹּאמֶר אֶל עֲבָדָיו עִבְרוּ לְפָנַי וְרֶוַח תָּשִׂימוּ בֵּין עֵדֶר וּבֵין עֵֽדֶר

    vayiten beyad avadayv eder eder levado vayomer el avadayv iveru lefanay verevach tasiymu beyn eder uveyn eder

    —

    18

    וַיְצַו אֶת הָרִאשׁוֹן לֵאמֹר כִּי יִֽפְגָּשְׁךָ עֵשָׂו אָחִי וִשְׁאֵֽלְךָ לֵאמֹר לְמִי אַתָּה וְאָנָה תֵלֵךְ וּלְמִי אֵלֶּה לְפָנֶֽיךָ

    vayetsav et harishon lemor kiy yifegashekha esav achiy visheelekha lemor lemiy atah veanah telekhe ulemiy eleh lefaneykha

    —

    19

    וְאָֽמַרְתָּ לְעַבְדְּךָ לְיַעֲקֹב מִנְחָה הִוא שְׁלוּחָה לַֽאדֹנִי לְעֵשָׂו וְהִנֵּה גַם הוּא אַחֲרֵֽינוּ

    veamareta leavedekha leyaaqov minechah hiv sheluchah ladoniy leesav vehineh gam hu achareynu

    —

    20

    וַיְצַו גַּם אֶת הַשֵּׁנִי גַּם אֶת הַשְּׁלִישִׁי גַּם אֶת כָּל הַהֹלְכִים אַחֲרֵי הָעֲדָרִים לֵאמֹר כַּדָּבָר הַזֶּה תְּדַבְּרוּן אֶל עֵשָׂו בְּמֹצַאֲכֶם אֹתֽוֹ

    vayetsav gam et hasheniy gam et hasheliyshiy gam et kal haholekhiym acharey haadariym lemor kadavar hazeh tedaberun el esav bemotsaakhem oto

    —

    21

    וַאֲמַרְתֶּם גַּם הִנֵּה עַבְדְּךָ יַעֲקֹב אַחֲרֵינוּ כִּֽי אָמַר אֲכַפְּרָה פָנָיו בַּמִּנְחָה הַהֹלֶכֶת לְפָנָי וְאַחֲרֵי כֵן אֶרְאֶה פָנָיו אוּלַי יִשָּׂא פָנָֽי

    vaamaretem gam hineh avedekha yaaqov achareynu kiy amar akhaperah fanayv baminechah haholekhet lefanay veacharey khen ereeh fanayv ulay yisa fanay

    —

    22

    וַתַּעֲבֹר הַמִּנְחָה עַל פָּנָיו וְהוּא לָן בַּלַּֽיְלָה הַהוּא בַּֽמַּחֲנֶֽה

    vataavor haminechah al panayv vehu lan balayelah hahu bamachaneh

    —

    23

    וַיָּקָם בַּלַּיְלָה הוּא וַיִּקַּח אֶת שְׁתֵּי נָשָׁיו וְאֶת שְׁתֵּי שִׁפְחֹתָיו וְאֶת אַחַד עָשָׂר יְלָדָיו וַֽיַּעֲבֹר אֵת מַעֲבַר יַבֹּֽק

    vayaqam balayelah hu vayiqach et shetey nashayv veet shetey shifechotayv veet achad asar yeladayv vayaavor et maavar yaboq

    —

    24

    וַיִּקָּחֵם וַיַּֽעֲבִרֵם אֶת הַנָּחַל וַֽיַּעֲבֵר אֶת אֲשֶׁר לוֹ

    vayiqachem vayaavirem et hanachal vayaaver et asher lo

    —

    25

    וַיִּוָּתֵר יַעֲקֹב לְבַדּוֹ וַיֵּאָבֵק אִישׁ עִמּוֹ עַד עֲלוֹת הַשָּֽׁחַר

    vayivater yaaqov levado vayeaveq iysh imo ad alot hashachar

    —

    26

    וַיַּרְא כִּי לֹא יָכֹל לוֹ וַיִּגַּע בְּכַף יְרֵכוֹ וַתֵּקַע כַּף יֶרֶךְ יַעֲקֹב בְּהֵֽאָבְקוֹ עִמּֽוֹ

    vayare kiy lo yakhol lo vayiga bekhaf yerekho vateqa kaf yerekhe yaaqov beheaveqo imo

    —

    27

    וַיֹּאמֶר שַׁלְּחֵנִי כִּי עָלָה הַשָּׁחַר וַיֹּאמֶר לֹא אֲשַֽׁלֵּחֲךָ כִּי אִם בֵּרַכְתָּֽנִי

    vayomer shalecheniy kiy alah hashachar vayomer lo ashalechakha kiy im berakhetaniy

    —

    28

    וַיֹּאמֶר אֵלָיו מַה שְּׁמֶךָ וַיֹּאמֶר יַעֲקֹֽב

    vayomer elayv mah shemekha vayomer yaaqov

    —

    29

    וַיֹּאמֶר לֹא יַעֲקֹב יֵאָמֵר עוֹד שִׁמְךָ כִּי אִם יִשְׂרָאֵל כִּֽי שָׂרִיתָ עִם אֱלֹהִים וְעִם אֲנָשִׁים וַתּוּכָֽל

    vayomer lo yaaqov yeamer od shimekha kiy im yiserael kiy sariyta im elohiym veim anashiym vatukhal

    —

    30

    וַיִּשְׁאַל יַעֲקֹב וַיֹּאמֶר הַגִּֽידָה נָּא שְׁמֶךָ וַיֹּאמֶר לָמָּה זֶּה תִּשְׁאַל לִשְׁמִי וַיְבָרֶךְ אֹתוֹ שָֽׁם

    vayisheal yaaqov vayomer hagiydah na shemekha vayomer lamah zeh tisheal lishemiy vayevarekhe oto sham

    —

    31

    וַיִּקְרָא יַעֲקֹב שֵׁם הַמָּקוֹם פְּנִיאֵל כִּֽי רָאִיתִי אֱלֹהִים פָּנִים אֶל פָּנִים וַתִּנָּצֵל נַפְשִֽׁי

    vayiqera yaaqov shem hamaqom peniyel kiy raiytiy elohiym paniym el paniym vatinatsel nafeshiy

    —

    32

    וַיִּֽזְרַֽח לוֹ הַשֶּׁמֶשׁ כַּאֲשֶׁר עָבַר אֶת פְּנוּאֵל וְהוּא צֹלֵעַ עַל יְרֵכֽוֹ

    vayizerach lo hashemesh kaasher avar et penuel vehu tsolea al yerekho

    —

    33

    עַל כֵּן לֹֽא יֹאכְלוּ בְנֵֽי יִשְׂרָאֵל אֶת גִּיד הַנָּשֶׁה אֲשֶׁר עַל כַּף הַיָּרֵךְ עַד הַיּוֹם הַזֶּה כִּי נָגַע בְּכַף יֶרֶךְ יַעֲקֹב בְּגִיד הַנָּשֶֽׁה

    al ken lo yokhelu veney yiserael et giyd hanasheh asher al kaf hayarekhe ad hayom hazeh kiy naga bekhaf yerekhe yaaqov begiyd hanasheh
